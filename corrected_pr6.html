import hashlib

def make_seed(last: str, first: str, number: int) -> int:
    # ФИО + номер -> стабильный seed 0..99999
    key = f"{last.strip().lower()}|{first.strip().lower()}|{int(number)}"
    return int(hashlib.sha256(key.encode("utf-8")).hexdigest(), 16) % 100_000

SEED = make_seed(last_name, first_name, list_number)

print("Студент:", last_name, first_name, f"(№{list_number})")
print("Ваш персональный SEED:", SEED)
print("Запишите SEED в итоговый отчёт — он подтверждает вариант.")
Как это работает. hashlib.sha256 переводит строку с ФИО и номером в большое число, затем берётся остаток от деления на 100 000. Одинаковые ФИО и номер всегда дают один и тот же SEED, разные — почти всегда разный. Весь датасет ниже зависит только от этого числа.

#@title Создание персонального датасета (нажмите ▶️) { display-mode: "form" }

def load_it_tickets(seed: int, n: int = 1800) -> "pd.DataFrame":
    """Индивидуальный «грязный» журнал IT-обращений по seed."""
    rng = np.random.default_rng(seed)

    opened_ts = rng.integers(1_735_689_600, 1_751_241_600, size=n)
    department = rng.choice(
        ["Support", "Backend", "DevOps", "Security", "Data"],
        size=n, p=[0.32, 0.22, 0.18, 0.14, 0.14],
    )
    product = rng.choice(
        ["Cloud", "CRM", "Billing", "Auth", "Analytics"], size=n
    )
    priority = rng.choice(
        ["Low", "Medium", "High", "Critical"], size=n, p=[0.40, 0.30, 0.20, 0.10]
    )
    channel = rng.choice(["Email", "Slack", "Portal", "Phone"], size=n)
    assignee = rng.choice(["Junior", "Middle", "Senior"], size=n, p=[0.35, 0.45, 0.20])
    tier = rng.choice(["Free", "Pro", "Enterprise"], size=n, p=[0.45, 0.40, 0.15])
    cpu_pct = np.clip(np.round(rng.normal(48, 18, size=n), 1), 1, 99)
    memory_mb = np.clip(np.round(rng.normal(2200, 850, size=n), 0), 256, None)
    n_reopens = rng.poisson(0.7, size=n)
    comments = rng.poisson(4.0, size=n) + 1
    resolve_hours = np.clip(np.round(rng.lognormal(1.55, 0.75, size=n), 2), 0.2, 240)

    logit = (
        -1.4
        + 1.4 * (priority == "Critical")
        + 0.7 * (priority == "High")
        - 0.4 * (priority == "Low")
        + 0.035 * resolve_hours
        + 0.018 * cpu_pct
        + 0.55 * (department == "Security")
        + 0.35 * (n_reopens >= 2)
        - 0.45 * (assignee == "Senior")
        - 0.30 * (tier == "Enterprise")
    )
    sla = (rng.uniform(size=n) < 1 / (1 + np.exp(-logit))).astype(int)

    df = pd.DataFrame({
        "ticket_id": np.arange(10_000, 10_000 + n),
        "opened_ts": opened_ts,
        "department": department,
        "product": product,
        "priority": priority,
        "channel": channel,
        "assignee_level": assignee,
        "client_tier": tier,
        "cpu_pct": cpu_pct,
        "memory_mb": memory_mb,
        "n_reopens": n_reopens,
        "comments": comments,
        "resolve_hours": resolve_hours,
        "sla_breached": sla,
    })

    df.loc[rng.choice(n, int(0.06 * n), replace=False), "resolve_hours"] = np.nan
    df.loc[rng.choice(n, int(0.04 * n), replace=False), "cpu_pct"] = np.nan
    df.loc[rng.choice(n, int(0.03 * n), replace=False), "comments"] = np.nan
    df.loc[rng.choice(n, 10, replace=False), "resolve_hours"] *= 9
    df.loc[rng.choice(n, 8, replace=False), "memory_mb"] *= 7
    ridx = rng.choice(n, int(0.12 * n), replace=False)
    df.loc[ridx, "department"] = df.loc[ridx, "department"].str.upper()
    pidx = rng.choice(n, int(0.08 * n), replace=False)
    df.loc[pidx, "product"] = "  " + df.loc[pidx, "product"].str.lower() + " "
    df = pd.concat([df, df.sample(int(0.025 * n), random_state=seed)], ignore_index=True)
    return df.sample(frac=1, random_state=seed).reset_index(drop=True)

print("Функция load_it_tickets готова.")

Ответы:



Groupby делает длинную таблицу (все данные идут вниз в столбик), а pivot_table делает двумерную сетку. Groupby удобнее для кода, дальнейшей обработки и графиков. Pivot_table удобнее для финальных отчетов, когда нужно посмотреть данные.



Применяет разные функции к разным столбцам за один шаг например считает mean для одного столбца и sum для другого, чтобы не делать лишних вычислений.



Потому что Unix-секунды это просто одно число. Месяцы имеют разное количество дней и секунд еще есть високосные года, поэтому перевести секунды в месяц простым делением невозможно.

Контрольные вопросы к этапу 5:


Когда для пропусков берут медиану, а когда среднее?

Что такое One-Hot (get_dummies) и почему отделы нельзя просто заменить числами 1, 2, 3…?

Зачем масштабировать признаки перед линейными моделями и kNN и почему дереву это обычно не нужно?

Почему ticket_id нельзя подавать в модель как обычный признак?
Медиану если распределение скошено или есть выбросы так как она устойчива, тогда как среднее тянется хвостом. Среднее уместно для симметричных данных без выбросов, когда нужен именно средний уровень.



One-Hot для каждой категории создаёт отдельный столбец. Например, если есть категория department=Security, появится столбец department_Security, где будет стоять 1, а в остальных столбцах для этой категории 0. Если же просто присвоить категориям числовые значения, получится ложный порядок и неверные расстояния между ними. Например, это будет означать, что категория Security в четыре раза больше Support или находится ровно посередине между двумя другими. Линейная модель может начать искать несуществующие закономерности, а дерево решений строить разделения, которых на самом деле нет в данных.



Линейные модели чувствительны к масштабу признаков. Например, memory_mb измеряется тысячами, а cpu_pct — десятками. Если не привести их к одному масштабу, признак с большими значениями будет иметь больший вес. Это замедлит работу алгоритмов, основанных на градиентном спуске, и в случае kNN приведёт к тому, что один признак будет затмить остальные при расчёте расстояний. StandardScaler решает эту проблему, приводя все столбцы к среднему значению 0 и стандартному отклонению 1. Для деревьев решений масштабирование не требуется, так как они выбирают порог для каждого признака отдельно.



ticket_id - это идентификатор, а не свойство модель может заучить номер 11598 это нарушение, и на новых данных такое предсказание станет случайным особенно легко это делают деревья.
